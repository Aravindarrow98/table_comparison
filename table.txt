# 📄 extractor.py

```python
from docx import Document


def extract_tables_from_docx(path):
    doc = Document(path)
    tables = []
    for table in doc.tables:
        rows = []
        for row in table.rows:
            rows.append([cell.text.strip() for cell in row.cells])
        tables.append(rows)
    return tables


def extract_tables_from_txt(path):
    tables = []
    with open(path, "r", encoding="utf-8") as f:
        content = f.read()

    raw_tables = content.split("\n\n")

    for raw in raw_tables:
        rows = [line.split("|") for line in raw.split("\n") if line.strip()]
        if rows:
            tables.append(rows)

    return tables


def extract_tables(path):
    if path.endswith(".docx"):
        return extract_tables_from_docx(path)
    elif path.endswith(".txt"):
        return extract_tables_from_txt(path)
    return []
```

# 🧠 comparison_engine.py

Includes:

* Semantic matching (SentenceTransformer + FAISS)
* Column alignment scoring
* Cell-level diff output

```python
import numpy as np
import faiss
from sentence_transformers import SentenceTransformer
from extractor import extract_tables
from rapidfuzz import fuzz

model = SentenceTransformer("all-MiniLM-L6-v2")


def table_to_string(table):
    return " ".join([" ".join(row) for row in table])


def column_similarity(t1, t2):
    cols1 = list(zip(*t1)) if t1 else []
    cols2 = list(zip(*t2)) if t2 else []

    scores = []

    for c1 in cols1:
        c1_string = " ".join(c1)
        best = 0
        for c2 in cols2:
            c2_string = " ".join(c2)
            score = fuzz.ratio(c1_string, c2_string) / 100
            best = max(best, score)
        scores.append(best)

    return float(np.mean(scores)) if scores else 0


def build_faiss_index(embeddings):
    dim = embeddings.shape[1]
    index = faiss.IndexFlatL2(dim)
    index.add(embeddings)
    return index


def cell_level_diff(t1, t2):
    diff_matrix = []

    for r in range(max(len(t1), len(t2))):
        row_diff = []
        for c in range(max(len(t1[0]) if t1 else 0,
                           len(t2[0]) if t2 else 0)):

            val1 = t1[r][c] if r < len(t1) and c < len(t1[r]) else ""
            val2 = t2[r][c] if r < len(t2) and c < len(t2[r]) else ""

            row_diff.append({
                "file1": val1,
                "file2": val2,
                "match": val1 == val2
            })

        diff_matrix.append(row_diff)

    return diff_matrix


def enterprise_compare(path1, path2):

    tables1 = extract_tables(path1)
    tables2 = extract_tables(path2)

    strings1 = [table_to_string(t) for t in tables1]
    strings2 = [table_to_string(t) for t in tables2]

    emb1 = model.encode(strings1)
    emb2 = model.encode(strings2)

    index = build_faiss_index(np.array(emb2).astype("float32"))

    D, I = index.search(np.array(emb1).astype("float32"), k=1)

    results = []

    for i, idx in enumerate(I):
        j = idx[0]
        similarity = 1 / (1 + D[i][0])

        col_score = column_similarity(tables1[i], tables2[j])
        final_score = (similarity * 0.7) + (col_score * 0.3)

        if final_score > 0.75:
            results.append({
                "table1_index": i,
                "table2_index": j,
                "similarity": round(final_score, 2),
                "diff": cell_level_diff(tables1[i], tables2[j])
            })

    return {
        "matches": results
    }
```

# 🌐 app.py

```python
import os
from flask import Flask, render_template, request, redirect, url_for
from tasks import compare_task
from config import Config

app = Flask(__name__)
app.config.from_object(Config)

UPLOAD_FOLDER = "uploads"
os.makedirs(UPLOAD_FOLDER, exist_ok=True)


@app.route("/", methods=["GET"])
def index():
    return render_template("index.html")


@app.route("/compare", methods=["POST"])
def compare():
    file1 = request.files["file1"]
    file2 = request.files["file2"]

    path1 = os.path.join(UPLOAD_FOLDER, file1.filename)
    path2 = os.path.join(UPLOAD_FOLDER, file2.filename)

    file1.save(path1)
    file2.save(path2)

    task = compare_task.delay(path1, path2)

    return redirect(url_for("result", task_id=task.id))


@app.route("/result/<task_id>")
def result(task_id):
    task = compare_task.AsyncResult(task_id)

    if task.state != "SUCCESS":
        return "Processing..."

    return render_template("result.html", data=task.result)


if __name__ == "__main__":
    app.run(debug=True)
```

```javascript
const left = document.getElementById("leftPanel");
const right = document.getElementById("rightPanel");

function render() {

    data.matches.forEach(match => {

        const tableContainer1 = document.createElement("div");
        const tableContainer2 = document.createElement("div");

        match.diff.forEach(row => {

            const row1 = document.createElement("div");
            const row2 = document.createElement("div");

            row.forEach(cell => {

                const c1 = document.createElement("span");
                const c2 = document.createElement("span");

                c1.innerText = cell.file1;
                c2.innerText = cell.file2;

                c1.className = "cell " + (cell.match ? "match" : "diff");
                c2.className = "cell " + (cell.match ? "match" : "diff");

                row1.appendChild(c1);
                row2.appendChild(c2);
            });

            tableContainer1.appendChild(row1);
            tableContainer2.appendChild(row2);
        });

        left.appendChild(tableContainer1);
        right.appendChild(tableContainer2);
    });
}

// Scroll Sync
left.addEventListener("scroll", () => {
    right.scrollTop = left.scrollTop;
});

right.addEventListener("scroll", () => {
    left.scrollTop = right.scrollTop;
});

render();
```
